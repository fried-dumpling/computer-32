#define IO_BASE 0x4000000
#define KEY_IN 0
#define KEY_V 4
#define KEY_R 8
#define TERM_OUT 12

#define LISTSIZE 1000
#define BUFFSIZE 10
#define STACKSIZE 10

; <halt>
#macro HALT 0
##inf_loop:
    jmp.one zero.32B, ##inf_loop
#end

; <block>
#macro BLOCK 0
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
    nop
#end

; <multiply>
; out: #1 -> product
; in:  #2 -> multiplicand
; in:  #3 -> multiplier
#macro MULT 3
    set #1.32B, 0
##mult_loop:
    test #3.32B, one.32B
    jmp.zero zero.32B, ##mult_skip
    add #1.32B, #2.32B
##mult_skip:
    shiftri #3.32B, 1
    shiftli #2.32B, 1
    cmpi #3.32B, 0
    ijmp.zero zero.32B, ##mult_loop
#end
    
; <modulo>
; out: #1 -> remainder
; in:  #2 -> dividend
; in:  #3 -> divisor
; use: #4 -> tmp
#macro MOD 4
    set #1.S16H, 0x7FFF
    set #1.16L, 0xFFFF
    mov #4.32B, #3.32B
    cmpi #3.32B, 0
    jmp.zero zero.32B, ##mod_exit

##mod_init:
    cmp #3.32B, #1.32B
    jmp.pos zero.32B, ##mod_main
    cmp #3.32B, #2.32B
    jmp.pos zero.32B, ##mod_main
    shiftli #3.32B, 1
    jmp.one zero.32B, ##mod_init
    
##mod_main:
    shiftrc #3.32B, one.32B
    mov #1.32B, #2.32B

##mod_loop:
    cmp #1.32B, #3.32B
    jmp.neg zero.32B, ##mod_skip
    sub #1.32B, #3.32B
##mod_skip:
    shiftri #3.32B, 1
    cmp #4.32B, #3.32B
    ijmp.pos zero.32B, ##mod_loop
##mod_exit:
#end

.data 
    primeLen:   4, 0
    primeArr:   4*LISTSIZE, 0
    strBuff:    4*BUFFSIZE, 0
    4*STACKSIZE, 0xCCCCCCCC
    stackEnd:   

.text
    set stack.16L, stackEnd & 0xFFFF
    set stack.S16H, stackEnd >> 16
    set gen[24].16L, IO_BASE & 0xFFFF
    set gen[24].S16H, IO_BASE >> 16
    set gen[23].16L, primeArr & 0xFFFF
    set gen[23].S16H, primeArr >> 16

    set gen[1].32B, 'l'
    st.gen[24] gen[1].32B, TERM_OUT
    set gen[1].32B, 'e'
    st.gen[24] gen[1].32B, TERM_OUT
    set gen[1].32B, 'n'
    st.gen[24] gen[1].32B, TERM_OUT
    set gen[1].32B, ':'
    st.gen[24] gen[1].32B, TERM_OUT
    set gen[1].32B, ' '
    st.gen[24] gen[1].32B, TERM_OUT

    call zero.32B, func_getInput
    call zero.32B, func_convertInt
    st.zero gen[0].32B, primeLen
    
    mov gen[1].32B, gen[23].32B
    call zero.32B, func_getPrime

    ld.zero gen[2].32B, primeLen
    subi gen[2].32B, 1
    shiftli gen[2].32B, 2
    ld.gen[2] gen[1].32B, primeArr
    call zero.32B, func_convertStr
    call zero.32B, func_print

    wait



    ; <get prime> 
    ; mem:  mem[primeArr]
    ; in:  gen[0] -> len
    ; use: gen[1] -> primeArr
    ; use: gen[2] -> cur
    ; use: gen[3] -> test
    ; use: gen[4] -> pointer
    ; use: gen[5] -> square
    ; use: gen[6] -> tmp
    ; use: gen[7] -> tmp2
    ; use: gen[8] -> write
    ; use: gen[9] -> tmp3
func_getPrime:
    set gen[6].32B, 2
    st.gen[1] gen[6].32B, 0
    set gen[2].32B, 3
    mov gen[8].32B, gen[1].32B
    addi gen[8].32B, 4
    
getPrime_mainloop:
    mov gen[4].32B, gen[1].32B
    
getPrime_subloop:
    ld.gen[4] gen[3].32B, 0
    mov gen[6].32B, gen[3].32B

    MOD(gen[7], gen[2], gen[6], gen[9])
    cmpi gen[7].32B, 0
    jmp.zero zero.32B, getPrime_continue

    addi gen[4].32B, 4
    mov gen[6].32B, gen[3].32B
    MULT(gen[5], gen[6], gen[3])
    cmp gen[5].32B, gen[2].32B
    ijmp.pos zero.32B, getPrime_subloop

    st.gen[8] gen[2].32B, 0
    addi gen[8].32B, 4

    subi gen[0].32B, 1
    cmpi gen[0].32B, 0
    ijmp.pos zero.32B, getPrime_exit

getPrime_continue:
    addi gen[2].32B, 1
    jmp.one zero.32B, getPrime_mainloop

getPrime_exit:
    ret

    ; <get input>
    ; out: mem[strBuff]
    ; use: gen[1] -> counter
    ; use: gen[2] -> strBuff base
    ; use: gen[3] -> valid holder
    ; use: gen[4] -> char holder
func_getInput:
    set gen[1].32B, 1           ; gen[1](counter) = 1
    set gen[2].32B, strBuff     ; gen[2](base) = strBuff

getInput_mainloop:
    ; check valid
    block
    ld.gen[24] gen[3].32B, KEY_V        ; gen[3](valid) = mem[IO_BASE + KEY_V]
    cmp gen[3].32B, zero.32B            ; valid == 0
    jmp.zero zero.32B, getInput_skip    ; if !valid skip

    ; get input
    ld.gen[24] gen[4].32B, KEY_IN       ; char = mem[IO_BASE + KEY_IN]
    st.gen[24] full.32B, KEY_R          ; mem[IO_BASE + KEY_R] = 0xFFFFFFFF
    st.gen[24] gen[4].32B, TERM_OUT     ; mem[IO_BASE + TERM_OUT] = char
    
    cmpi gen[4].32B, '\n'               ; char == '\n'
    jmp.zero zero.32B, getInput_exit    ; if (char == '\n') break

    st.gen[2] gen[4].32B, 0     ; mem[base] = char
    addi gen[2].32B, 4          ; base += 4
    addi gen[1].32B, 1          ; count++
    cmpi gen[1].32B, BUFFSIZE   ; count == buffSize
    jmp.zero zero.32B, getInput_exit    ; if (count == buffSize) break
    
getInput_skip:
    jmp.one zero.32B, getInput_mainloop ; continue to loop

getInput_exit:
    set gen[4].32B, 0       ; char = 0
    st.gen[2] gen[4].32B, 0 ; mem[base] = char(0)
    ret

    ; <convert int>
    ; in: mem[strBuff]
    ; out: gen[0] -> number
    ; use: gen[1] -> char
    ; use: gen[2] -> base
    ; use: gen[3] -> tmp
func_convertInt:
    set gen[0].32B, 0           ; number = 0
    set gen[2].32B, strBuff    ; base = strBuff
    
convertInt_mainloop:
    ld.gen[2] gen[1].32B, 0     ; char = mem[base]
    cmp gen[1].32B, zero.32B    ; char == 0

    jmp.zero zero.32B, convertInt_exit  ; if (char == 0) break

    ;number *= 10
    mov gen[3].32B, gen[0].32B      ; gen[3] = gen[0]
    shiftli gen[0].32B, 1           ; gen[0] * 2
    shiftli gen[3].32B, 3           ; gen[3] * 8
    add gen[0].32B, gen[3].32B      ; gen[0] += gen[3]

    addi gen[2].32B, 4          ; base += 4
    subi gen[1].32B, 48         ; char -= 48
    add gen[0].32B, gen[1].32B  ; number += char
    jmp.one zero.32B, convertInt_mainloop   ; continue loop
    
convertInt_exit:
    ret

    ; <convert str>
    ; in: gen[1] -> number
    ; out: mem[strBuff] -> string
    ; use: gen[2] -> base
    ; use: gen[3] -> result
    ; use: gen[4] -> c3
    ; use: gen[5] -> tmp
    ; use: gen[6] -> mask
    ; use: gen[7] -> counter
    ; use: gen[8] -> sum
    ; use: gen[9] -> tmp2
func_convertStr:
    set gen[3].32B, 0       ; result = 0
    set gen[7].32B, 0       ; counter = 0

    set gen[4].S16H, 0x3333
    set gen[4].16L, 0x3333      ; c3 = 0x33333333
    set gen[6].S16H, 0x8888
    set gen[6].16L, 0x8888      ; mask = 0x88888888

convertStr_mainloop:
    mov gen[5].32B, gen[3].32B      ; tmp = result
    add gen[5].32B, gen[4].32B      ; tmp += 0x33333333
    bnd gen[5].32B, gen[6].32B      ; tmp &= 0x88888888

    shiftri gen[5].32B, 2           
    mov gen[9].32B, gen[5].32B      
    shiftri gen[5].32B, 1           
    bor gen[9].32B, gen[5].32B      ; tmp2 = (tmp >> 3)*3

    add gen[3].32B, gen[9].32B      ; result += tmp2

    shiftli gen[1].32B, 1       ; (gen[1] <<= 1)
    shiftlc gen[3].32B, one.32B ; (gen[3] <<= 1) + carry

    addi gen[7].32B, 1              ; counter++
    cmpi gen[7].32B, 32             ; counter == 32
    jmp.zero zero.32B, convertStr_exit  ; if (counter == 32) break

    jmp.one zero.32B, convertStr_mainloop   ; continue loop

convertStr_exit:
    ;st.zero gen[3].32B, bcdTest

    set gen[2].32B, strBuff ; base = strBuff
    set gen[7].32B, 0               ; counter = 0
    set gen[8].32B, 0               ; sum = 0

convertStr_charloop:
    mov gen[5].32B, gen[3].32B      ; tmp = result
    shiftri gen[5].32B, 28          ; tmp >>= 28
    add gen[8].32B, gen[5].32B      ; sum += tmp

    cmpi gen[8].32B, 0              ; sum == 0
    jmp.zero zero.32B, convertStr_skipchar  ; if (sum == 0) skip

    addi gen[5].32B, 48             ; tmp += 48
    st.gen[2] gen[5].32B, 0         ; mem[base] = tmp
    addi gen[2].32B, 4              ; base += 4

convertStr_skipchar:
    shiftli gen[3].32B, 4           ; result <<= 4
    addi gen[7].32B, 1              ; counter++
    cmpi gen[7].32B, 8             ; counter == 8
    ijmp.zero zero.32B, convertStr_charloop ; if (counter != 8) loop

    ; exit
    st.gen[2] zero.32B, 0           ; mem[base] = 0
    ret                             ; return

    ; <print>
    ; in: mem[strBuff] -> string
    ; use: gen[1] -> char
    ; use: gen[2] -> base
func_print: 
    set gen[2].32B, strBuff     ; base = strBuff

print_mainloop:
    ld.gen[2] gen[1].32B, 0             ; char = mem[base]
    cmpi gen[1].32B, 0                  ; char == 0
    jmp.zero zero.32B, print_exit       ; if (char == 0) break

    st.gen[24] gen[1].32B, TERM_OUT     ; mem[IO_BASE, TERM_OUT] = tmp
    addi gen[2].32B, 4                  ; base += 4
    jmp.one zero.32B, print_mainloop    ; continue loop

print_exit:
    ret
