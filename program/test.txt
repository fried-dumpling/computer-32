#macro HALT 0
##halt_loop:
    jmp.one zero.32B, ##halt_loop
#end

#macro ADD 3
    mov #1.32B, #2.32B
    add #1.32B, #3.32B
#end

.data
    A: 4, 3
    B: 4, 5
    C: 4, 0xFFFFFFFF
.text
    ld.zero gen[1].32B, A
    ld.zero gen[2].32B, B
    ADD(gen[0], gen[1], gen[2])
    st.zero gen[0].32B, C
    HALT()
